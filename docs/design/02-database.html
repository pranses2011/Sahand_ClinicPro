📌 قراردادهای طراحی:
  • PK: همه کلیدهای اصلی bigint IDENTITY(1,1)
  • Soft Delete: فیلد IsDeleted bit در همه جداول
  • Audit: فیلدهای CreatedAt, CreatedBy, UpdatedAt, UpdatedBy
  • Encryption: فیلدهای حساس بیماران با AES-256 رمزنگاری می‌شوند
  • Collation: SQL_Latin1_General_CP1_CI_AS با پشتیبانی nvarchar
  • تاریخ شمسی: در کنار datetime، فیلد string nvarchar(10) ذخیره می‌شود
  • Multi-tenancy: فیلد BranchId در جداول آماده برای چند شعبه‌ای
  • Row Version: فیلد RowVersion timestamp برای Concurrency Control

🎯 Base Entity (جد مشترک همه موجودیت‌ها)

public abstract class BaseEntity { public long Id { get; set; } public DateTime CreatedAt { get; set; } public long? CreatedBy { get; set; } public DateTime? UpdatedAt { get; set; } public long? UpdatedBy { get; set; } public bool IsDeleted { get; set; } public byte[] RowVersion { get; set; } public long? BranchId { get; set; } // برای Multi-Branch }

-- View: VW_PatientSummary CREATE VIEW VW_PatientSummary AS SELECT p.PatientId, p.MedicalRecordNo, p.FullName, p.NationalCode, p.Mobile, p.Age, ic.Name AS InsuranceName, (SELECT COUNT(*) FROM Visits WHERE PatientId=p.PatientId) AS TotalVisits, (SELECT MAX(VisitDate) FROM Visits WHERE PatientId=p.PatientId) AS LastVisit, (SELECT SUM(RemainingAmount) FROM Invoices WHERE PatientId=p.PatientId) AS Debt FROM Patients p LEFT JOIN InsuranceCompanies ic ON p.InsuranceCompanyId=ic.InsuranceCompanyId WHERE p.IsDeleted = 0; -- View: VW_DailyRevenue CREATE VIEW VW_DailyRevenue AS SELECT CAST(InvoiceDate AS DATE) AS Date, COUNT(*) AS InvoiceCount, SUM(TotalAmount) AS Total, SUM(PaidAmount) AS Paid, SUM(RemainingAmount) AS Remaining FROM Invoices WHERE IsDeleted = 0 GROUP BY CAST(InvoiceDate AS DATE); -- + VW_DoctorPerformance, VW_InventoryLow, VW_AppointmentToday, VW_InsuranceClaimSummary ...

🔍 ایندکس‌های کلیدی

-- Patients CREATE UNIQUE INDEX IX_Patients_NationalCode ON Patients(NationalCode) WHERE IsDeleted=0; CREATE INDEX IX_Patients_Mobile ON Patients(Mobile); CREATE INDEX IX_Patients_LastName ON Patients(LastName); -- Appointments CREATE INDEX IX_Appointments_Date_Doctor ON Appointments(AppointmentDate, DoctorId); CREATE INDEX IX_Appointments_Patient ON Appointments(PatientId); CREATE INDEX IX_Appointments_Status ON Appointments(Status) INCLUDE(AppointmentDate); -- Visits CREATE INDEX IX_Visits_Patient_Date ON Visits(PatientId, VisitDate DESC); CREATE INDEX IX_Visits_Doctor_Date ON Visits(DoctorId, VisitDate); -- Invoices CREATE INDEX IX_Invoices_Date ON Invoices(InvoiceDate DESC); CREATE INDEX IX_Invoices_Patient ON Invoices(PatientId); CREATE INDEX IX_Invoices_Status ON Invoices(PaymentStatus); -- AuditLogs CREATE INDEX IX_AuditLogs_User_Time ON AuditLogs(UserId, Timestamp DESC); CREATE INDEX IX_AuditLogs_Entity ON AuditLogs(EntityName, EntityId); -- Pharmacy CREATE INDEX IX_PharmacyInventory_Expiry ON PharmacyInventory(ExpiryDate); CREATE INDEX IX_PharmacyInventory_LowStock ON PharmacyInventory(Quantity) WHERE Quantity < ReorderLevel;
