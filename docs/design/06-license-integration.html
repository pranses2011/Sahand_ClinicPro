// در License Portal (Server-Side) - تولید فایل لایسنس public class LicenseGenerator { private readonly RSA _rsa; // Private Key (در Portal) public LicenseFile Generate(LicenseRequest req) { var license = new LicenseFile { LicenseKey = GenerateKey(), CustomerId = req.CustomerId, ClinicName = req.ClinicName, LicenseType = req.LicenseType, MaxUsers = req.MaxUsers, MaxDoctors = req.MaxDoctors, EnabledModules = req.EnabledModules, IssuedDate = DateTime.UtcNow, ExpiryDate = req.ExpiryDate, HardwareFingerprint = req.HardwareFingerprint, }; // Serialize to JSON (sorted keys for consistent hash) var json = JsonSerializer.Serialize(license, new JsonSerializerOptions { WriteIndented = false }); var dataBytes = Encoding.UTF8.GetBytes(json); // Sign with RSA Private Key (SHA-256) var signature = _rsa.SignData(dataBytes, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1); license.Signature = Convert.ToBase64String(signature); return license; } } // در Client (نرم‌افزار درمانگاه) - بررسی لایسنس public class LicenseVerifier { private readonly RSA _rsa; // Public Key (در Client - Embedded) public bool Verify(LicenseFile license) { // 1. Check Hardware Fingerprint var currentFingerprint = HardwareFingerprint.Generate(); if (license.HardwareFingerprint != currentFingerprint) return false; // سخت‌افزار تغییر کرده // 2. Check Expiry if (license.ExpiryDate.HasValue && license.ExpiryDate < DateTime.UtcNow) return false; // منقضی شده // 3. Verify RSA Signature var signature = Convert.FromBase64String(license.Signature); var licenseCopy = license.Clone(); licenseCopy.Signature = null; // حذف قبل از verify var json = JsonSerializer.Serialize(licenseCopy); var dataBytes = Encoding.UTF8.GetBytes(json); return _rsa.VerifyData(dataBytes, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1); } }

// Hangfire Job - اجرا هر ۲۴ ساعت public class LicenseHeartbeatJob { private readonly ILicenseClient _client; private readonly IUnitOfWork _uow; [AutomaticRetry(Attempts = 3)] public async Task ExecuteAsync() { var license = await _uow.Licenses.GetActiveAsync(); // 1. Check expiry locally if (license.ExpiryDate.HasValue) { var daysLeft = (license.ExpiryDate.Value - DateTime.UtcNow).Days; if (daysLeft <= 30 && daysLeft > 0) await NotifyAdmin($"لایسنس تا {daysLeft} روز دیگر منقضی می‌شود"); } // 2. Online validation (if internet available) try { var result = await _client.ValidateAsync(license.LicenseKey); if (!result.IsValid) { // فروشنده لایسنس را revoke کرده license.Status = LicenseStatus.Suspended; await _uow.SaveChangesAsync(); await NotifyAdmin("لایسنس توسط فروشنده معلق شده است"); } license.LastValidatedAt = DateTime.UtcNow; await _uow.SaveChangesAsync(); } catch (HttpRequestException) { // آفلاین - اگر بیش از ۳۰ روز offline، لایسنس invalid می‌شود var daysOffline = (DateTime.UtcNow - license.LastValidatedAt).Days; if (daysOffline > 30 && !license.IsOfflineAllowed) throw new LicenseException("نیاز به اتصال به سرور"); } } }

🛡️ Middleware بررسی لایسنس (API)

public class LicenseValidationMiddleware { private readonly RequestDelegate _next; private readonly ILicenseService _license; public async Task InvokeAsync(HttpContext context) { // Skip for license endpoints if (context.Request.Path.StartsWithSegments("/api/license")) { await _next(context); return; } // Check license validity var status = await _license.GetStatusAsync(); if (!status.IsValid) { context.Response.StatusCode = 403; await context.Response.WriteAsJsonAsync(new { Error = "License Invalid", Reason = status.Reason, Code = "LICENSE_INVALID" }); return; } // Check concurrent users var activeUsers = await _license.GetActiveUserCountAsync(); if (activeUsers > status.MaxUsers) { context.Response.StatusCode = 403; await context.Response.WriteAsJsonAsync(new { Error = "User Limit Exceeded", Code = "USER_LIMIT_EXCEEDED" }); return; } // Check module access var module = ExtractModuleFromPath(context.Request.Path); if (module != null && !status.EnabledModules.Contains(module)) { context.Response.StatusCode = 403; await context.Response.WriteAsJsonAsync(new { Error = $"Module '{module}' not licensed", Code = "MODULE_NOT_LICENSED" }); return; } await _next(context); } }

🔒 اقدامات ضد تقلب (Anti-Piracy)

🧩 Code Obfuscation

استفاده از Dotfuscator یا ConfuserEx برای درهم‌ریختن کد و دشوار کردن Reverse Engineering

🔐 Signed Assemblies

همه DLLها با Strong Name امضا می‌شوند تا جلوگیری از تعویض شوند

🖥️ Hardware Binding

لایسنس به سخت‌افزار خاص گره خورده - تعویض سخت‌افزار نیاز به reactivation دارد

⏰ Time Tampering Detection

بررسی تغییرات زمان سیستم. اگر زمان به عقب برگردانده شد، لایسنس غیرفعال می‌شود

📡 Online Heartbeat

چک آنلاین دوره‌ای. فروشنده می‌تواند لایسنس را از راه دور Revoke کند

🧬 Integrity Check

در زمان اجرا، hash فایل‌های اصلی چک می‌شود تا تغییر پیدا نکرده باشند

🏛️ بخش دوم: یکپارچه‌سازی با سامانه‌های دولتی ایران

🎯 استراتژی یکپارچه‌سازی: هر سامانه دولتی دارای Service Interface جداگانه است (Strategy Pattern). این باعث می‌شود:
  • تغییر API سامانه‌ها بدون تاثیر بر بقیه کد
  • Mock شدن آسان برای تست
  • اضافه کردن سامانه جدید بدون تغییر کد موجود
  • مدیریت Retry و Circuit Breaker با Polly

📅 الگوریتم ۱: محاسبه Slot‌های نوبت‌دهی موجود

با توجه به برنامه پزشک، نوبت‌های رزرو شده و غیبت‌ها، لیست ساعت‌های آزاد را محاسبه می‌کند.

  1. دریافت ورودی: DoctorId, Date, SlotDuration (مثلاً ۱۵ دقیقه)
  2. بارگذاری DoctorSchedule برای روز هفته (مثلاً دوشنبه)
  3. بررسی DoctorAbsences برای تاریخ خاص (مرخصی، جلسه، ...)
  4. ساخت لیست کامل Slot‌های نظری (مثلاً از ۸:۰۰ تا ۱۴:۰۰ هر ۱۵ دقیقه)
  5. حذف Slot‌هایی که در BreakTime هستند
  6. دریافت نوبت‌های رزرو شده (Appointments) برای این تاریخ و پزشک
  7. حذف Slot‌های رزرو شده از لیست
  8. برگشت لیست Slot‌های آزاد به همراه ظرفیت باقیمانده
⏱️ O(n) 📦 Cache 1 hour 🔒 Thread-safe
💰 الگوریتم ۲: محاسبه سهم بیمه و بیمار (فرانشیز)

محاسبه دقیق سهم بیمه اصلی، بیمه تکمیلی و بیمار بر اساس تعرفه و قوانین

  1. دریافت ورودی: ServiceId, Quantity, InsuranceCompanyId, SupplementaryInsuranceId
  2. دریافت تعرفه خدمت از Tariffs برای InsuranceCompanyId و سال جاری
  3. اگر تعرفه بیمه موجود نبود، تعرفه آزاد استفاده شود
  4. محاسبه قیمت کل: Price = Tariff.Price × Quantity
  5. اگر IsK فعال، محاسبه با فرمول K: Price = Tariff.KValue × GlobalK × Quantity
  6. محاسبه سهم بیمه اصلی: InsuranceShare = Price × (Tariff.InsuranceShare / 100)
  7. اعمال سقف پوشش بیمه: if InsuranceShare > Ceiling then InsuranceShare = Ceiling
  8. محاسبه سهم باقیمانده: Remaining = Price - InsuranceShare
  9. اگر بیمه تکمیلی وجود دارد: SuppShare = Remaining × SuppPercent
  10. محاسبه سهم نهایی بیمار: PatientShare = Remaining - SuppShare
  11. اعمال تخفیف نهایی (اگر وجود دارد)
  12. برگشت ساختار کامل محاسبات
⏱️ O(1) 💾 نیاز به دیتابیس
⚠️ الگوریتم ۳: بررسی تداخل دارویی (Drug Interaction Check)

هنگام نسخه‌نویسی، تداخلات بین داروهای جدید و قبلی را شناسایی می‌کند.

  1. دریافت لیست داروهای نسخه فعلی: CurrentMedications[]
  2. دریافت داروهای فعلی بیمار (قبل‌ها): PatientActiveMedications[]
  3. ترکیب هر دو لیست: AllMeds = Current ∪ PatientActive
  4. برای هر جفت دارو (Med1, Med2) در AllMeds:
  5. جستجو در DrugInteractions: WHERE (Med1Id = @m1 AND Med2Id = @m2) OR (Med1Id = @m2 AND Med2Id = @m1)
  6. اگر تداخل پیدا شد، به لیست نتایج اضافه کن
  7. مرتب‌سازی نتایج بر اساس Severity (Severe → Moderate → Mild)
  8. بررسی آلرژی‌های بیمار (PatientAllergies) با مواد تشکیل‌دهنده داروها
  9. اگر تداخل Severe یا آلرژی وجود دارد، نیاز به تایید پزشک
  10. بازگشت گزارش کامل تداخلات به UI
⏱️ O(n²) worst case 📦 Index در DrugInteractions 🧠 Cache
📒 الگوریتم ۴: تولید خودکار سند حسابداری دوبل

هر رویداد مالی (فاکتور، پرداخت، حقوق) به صورت خودکار سند حسابداری دوبل تولید می‌کند.

  1. دریافت ورودی: EventType (Invoice/Payment/Salary), EventData
  2. بارگذاری قوانین حسابداری بر اساس EventType از جدول Rules
  3. برای هر قانون:
    • تعیین DebitAccountId و CreditAccountId
    • محاسبه Amount بر اساس فرمول (معمولاً Direct Mapping)
  4. ایجاد JournalEntry اصلی با:
    • EntryNumber = Auto-generated
    • EntryDate = Today
    • Description = از Template
    • ReferenceType = Invoice/Payment/...
    • ReferenceId = ID رویداد
  5. ایجاد JournalEntryLines برای هر طرف:
    • سطر بدهکار: AccountId, Debit=Amount, Credit=0
    • سطر بستانکار: AccountId, Debit=0, Credit=Amount
  6. بررسی قانون طلایی: SUM(Debit) == SUM(Credit)
  7. اگر نامتوازن، throw AccountingException
  8. ثبت در Transaction و SaveChanges
  9. به‌روزرسانی CurrentBalance حساب‌های درگیر
  10. برگشت EntryId به فراخواننده

📊 الگوریتم ۵: مدیریت صف انتظار پویا

مدیریت هوشمند صف بیماران با در نظر گرفتن اولویت، نوبت رزرو شده و تاخیر پزشک

  1. دریافت لیست پذیرش‌های امروز برای پزشک DoctorId
  2. برای هر پذیرش، محاسبه Priority Score:
    • بیماران اورژانسی: +۱۰۰۰
    • VIP: +۵۰۰
    • بیمه کارمند: +۱۰۰
    • تاخیر زمان نوبت: +۱۰ به ازای هر ۱۵ دقیقه
    • مراجعه مجدد: +۵۰
  3. مرتب‌سازی بر اساس: (ArrivedAt ASC, PriorityScore DESC)
  4. تخصیص شماره صف پویا
  5. محاسبه زمان تخمینی ویزیت:
    • EstimatedWait = (QueuePosition × AvgVisitDuration) - ElapsedTime
  6. به‌روزرسانی Real-time با SignalR به کلاینت‌ها
  7. اگر بیمار بیش از ۳۰ دقیقه در صف است، نوتیفیکیشن به منشی
⏱️ O(n log n) 📡 SignalR 🔄 Real-time
📱 الگوریتم ۶: ارسال خودکار پیامک یادآوری نوبت

Hangfire Job روزانه برای ارسال پیامک یادآوری به بیماران فردا

  1. اجرای Job هر روز ساعت ۱۰ صبح
  2. دریافت لیست نوبت‌های فردا با وضعیت Scheduled
  3. فیلتر نوبت‌هایی که قبلاً پیامک ارسال نشده (ReminderSent = false)
  4. برای هر نوبت:
    • بارگذاری Template پیامک "یادآوری نوبت"
    • جایگزینی متغیرها: {PatientName}, {DoctorName}, {Date}, {Time}, {Address}
    • افزودن به SmsQueue با اولویت متوسط
  5. Hangfire Worker دیگر هر ۱ دقیقه SmsQueue را پردازش می‌کند
  6. انتخاب پنل پیامک پیش‌فرض
  7. ارسال با Retry Policy (۳ بار)
  8. به‌روزرسانی SmsMessages: Status, SentAt, Cost, ProviderMessageId
  9. به‌روزرسانی Appointments.ReminderSent = true
  10. پیگیری Delivery Status بعد از ۵ دقیقه
⏰ Hangfire Daily 🔁 Retry 3x 📊 Queued
💾 الگوریتم ۷: بکاپ‌گیری خودکار Multi-Tier

سیستم بکاپ ۳ سطحی: Full روزانه، Differential ۶ ساعته، Transaction Log هر ساعت

  1. Hangfire Jobs با زمان‌بندی:
    • Full Backup: هر روز ۳:۰۰ بامداد
    • Differential: هر ۶ ساعت
    • Transaction Log: هر ۱ ساعت
  2. برای Full Backup:
    • اجرای T-SQL: BACKUP DATABASE ... TO DISK WITH COMPRESSION, CHECKSUM
    • نام فایل: ClinicPro_Full_{Timestamp}.bak
    • رمزنگاری فایل با AES-256 و کلید اختصاصی
  3. آپلود به ذخیره‌سازی ثانویه (در صورت فعال بودن):
    • Cloud: Azure Blob / AWS S3 / آروان‌کلاد
    • FTP: به سرور دیگر
    • External Drive
  4. سیاست نگهداری (Retention Policy):
    • Full: ۳۰ روز اخیر + ۱ بکاپ هر ماه (۱۲ ماه)
    • Diff: ۷ روز اخیر
    • Log: ۴۸ ساعت اخیر
  5. حذف خودکار بکاپ‌های قدیمی
  6. تست Integrity هر بکاپ: RESTORE VERIFYONLY
  7. ثبت در BackupHistory
  8. در صورت خطا: ارسال نوتیفیکیشن اضطراری
  9. گزارش ماهانه بکاپ به مدیر سیستم

🔍 الگوریتم ۸: جستجوی سریع بیماران (Full-Text + Fuzzy)

جستجوی هوشمند با تحمل خطا در تایپ، فارسی/عربی، و اولویت‌دهی هوشمند

  1. دریافت ورودی جستجو: SearchTerm (می‌تواند نام، کد ملی، موبایل، شماره پرونده باشد)
  2. تشخیص نوع ورودی با Regex:
    • ^\d{10}$ → کد ملی (جستجوی دقیق)
    • ^09\d{9}$ → موبایل (جستجوی دقیق)
    • ^\d{5,8}$ → شماره پرونده
    • سایر → نام (Fuzzy)
  3. نرمال‌سازی متن:
    • تبدیل ی/ك عربی به ی/ک فارسی
    • حذف فاصله‌های اضافی
    • تبدیل اعداد فارسی/عربی به انگلیسی
  4. برای نام: استفاده از SQL Server Full-Text Index با CONTAINS و FORMSOF(INFLECTIONAL)
  5. اضافه کردن Fuzzy Search با SOUNDEX یا DIFFERENCE برای غلط‌های املایی
  6. اولویت‌بندی نتایج:
    • +۱۰۰ اگر نام کامل Match
    • +۵۰ اگر نام خانوادگی Match
    • +۳۰ اگر نام Match
    • +۱۰۰ اگر مراجعه در ماه اخیر
  7. Pagination با limit ۵۰ نتیجه
  8. Highlight کردن متن Match شده در UI
  9. Cache نتیجه برای ۵ دقیقه
⏱️ O(log n) با Index 🔎 Full-Text 💡 Fuzzy
📊 الگوریتم ۹: محاسبه کارانه پزشک

محاسبه خودکار سهم پزشک از ویزیت‌ها، آزمایش‌ها، نسخه‌ها و اعمال پاداش/کسر

  1. دریافت ورودی: DoctorId, PeriodStart, PeriodEnd
  2. دریافت تمام فاکتورهای پرداخت‌شده (PaymentStatus = Paid) این پزشک در این دوره
  3. برای هر InvoiceItem از این پزشک:
    • محاسبه DoctorShare = (Price × DoctorSharePercent) / 100
    • افزودن به GrossAmount
  4. کسر مالیات (Tax = GrossAmount × TaxRate)
  5. کسر بیمه (در صورت وجود)
  6. کسر وام‌ها و پیش‌پرداخت‌ها
  7. افزودن پاداش (در صورت وجود)
  8. محاسبه نهایی: NetAmount = Gross - Tax - Insurance - Loans + Bonus
  9. ذخیره در DoctorPayments
  10. تولید سند حسابداری:
    • بدهکار: حساب هزینه کارانه پزشکان
    • بستانکار: حساب بانک / پرداختنی
  11. تولید PDF فیش حقوقی
  12. ارسال SMS اطلاع واریز
⏱️ O(n) 📒 Auto Journal 📊 Monthly
⚠️ الگوریتم ۱۰: هشدار موجودی کم داروخانه

تحلیل روزانه موجودی و پیش‌بینی نیاز سفارش بر اساس مصرف تاریخی

  1. Hangfire Job روزانه ساعت ۸ صبح
  2. برای هر آیتم در PharmacyInventory:
    • بررسی موجودی فعلی: Quantity
    • محاسبه میانگین مصرف ۳۰ روز گذشته: AvgDailyUsage
    • پیش‌بینی مدت باقیمانده: DaysLeft = Quantity / AvgDailyUsage
  3. طبقه‌بندی:
    • Critical (قرمز): Quantity < MinStock یا DaysLeft < 3
    • Low (زرد): Quantity < ReorderLevel یا DaysLeft < 7
    • OK (سبز): سایر موارد
  4. بررسی تاریخ انقضا:
    • اگر ExpiryDate < Now + 30 days → هشدار
    • اگر ExpiryDate < Now → بلاک تحویل
  5. تولید سفارش خرید پیشنهادی برای آیتم‌های Critical/Low:
    • SuggestedOrder = MaxStock - Quantity
    • انتخاب تامین‌کننده با آخرین خرید / بهترین قیمت
  6. ارسال گزارش به داروساز و مدیر
  7. نمایش در داشبورد با تعداد موارد Critical/Low
⏰ Hangfire Daily 📊 Predictive 🔔 Alerts
🖨️ الگوریتم ۱۱: تولید شماره‌های یکتا (Invoice, MRN, ...)

تولید Sequential شماره‌های یکتا با پیشوند و Thread-Safe

  1. تعریف Pattern برای هر نوع شماره:
    • فاکتور: INV-{YYYY}-{SeqNo:D6} مثلاً INV-2024-000123
    • پرونده: MRN-{SeqNo:D6} مثلاً MRN-010234
    • نسخه: RX-{YYYYMM}-{SeqNo:D5}
    • سند حسابداری: JE-{YYYY}-{SeqNo:D5}
  2. استفاده از SQL Server Sequence (کارآمدترین روش):
    • CREATE SEQUENCE InvoiceSeq START WITH 1 INCREMENT BY 1
    • فراخوانی: SELECT NEXT VALUE FOR InvoiceSeq
  3. فرمت نهایی با C# String Format
  4. بررسی تکراری نبودن قبل از Insert
  5. در صورت تکراری بودن (edge case)، retry با شماره جدید
  6. ثبت در Audit Log
⏱️ O(1) 🔒 Thread-safe ⚡ SQL Sequence
